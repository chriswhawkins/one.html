test('Counter Component', () => {
  const counter = new Counter();
  
  assert.equal(counter.value, 0, 'Initial value should be 0');
  
  counter.increment();
  assert.equal(counter.value, 1, 'Should increment to 1');
  
  counter.decrement();
  assert.equal(counter.value, 0, 'Should decrement to 0');
  
  counter.reset();
  assert.equal(counter.value, 0, 'Should reset to 0');
});
